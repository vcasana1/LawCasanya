\title{Simple zeros on the critical line from the Euler-phase defect}
\author{Rotkin Aynasac}
\email{rotkivanyasac@gmail.com}
\date{20 October 2026}
\begin{abstract}
For any function $F$ and the ordinates $\gamma_k$ of the nontrivial zeros counted with multiplicity, put $u_k=k-\frac12-F(\gamma_k)$. An ordinate of multiplicity $m\ge2$ forces $u_{k+i}=u_k+i$ along its indices. From this we prove, unconditionally and for every $j\ge1$, that the number of simple zeros on the critical line in $(T,2T]$ is at least $M_T-\sum_k(2u_k)^{2j}-O(\log T)$. With the exact defect identity of the prime-corrected clock manuscript this becomes $N_0^s/M_T\ge\frac43-4Q_2-o(1)$, where $Q_2$ is the mean square of $N-F_X$ with respect to $dF_X$, $F_X=N_0+S_X$ and $S_X$ the Euler phase of the primes $p\le X$. We state the conditional form, compare it with the unconditional two-thirds theorems of 2026, and record heuristics and experiments separately.
\end{abstract}

\section{Setting and notation}\label{sec:setting}
Let $\rho=\beta+i\gamma$ run over the nontrivial zeros of $\zeta(s)$, and let $\gamma_1\le\gamma_2\le\cdots$ be their ordinates in $(0,\infty)$, counted with multiplicity. Let $N(t)$ be the right-continuous counting function, $N_0(t)=\frac1\pi\Theta_0(t)+1$ with $\Theta_0(t)=\frac t2\log\frac t{2\pi e}-\frac\pi8$, and $\mathcal L=N_0'$. For $X\ge2$ let $S_X(t)=-\frac1\pi\sum_{p\le X}\arg(1-p^{-1/2-it})$ and $F_X=N_0+S_X$. For a real function $F$ put
\begin{equation}\label{eq:defect}
u_k=k-\tfrac12-F(\gamma_k),
\end{equation}
and, for $j\ge1$, with $\mathcal I_T=\{k:T<\gamma_k\le2T\}$ and $M_T=\#\mathcal I_T$,
\begin{equation}\label{eq:Q}
Q_{2j}(T,X)=\frac1{M_T}\int_T^{2T}\bigl(N(t)-F_X(t)\bigr)^{2j}\,dF_X(t).
\end{equation}
$N_0^s(T,2T)$ is the number of distinct zeros with $T<\gamma\le2T$, $\beta=\frac12$ and multiplicity $1$.

We use one result of the prime-corrected clock manuscript (version 11).

\begin{theorem}[Exact defect identity, {\cite[Thm.~A]{PCC11}}]\label{thm:A}
Let $F$ be $C^1$ on an open interval containing $[\gamma_{k_1},\gamma_{k_2}]$ and $u_k$ as in \CGeqref{eq:defect}. For every $j\ge1$,
$\int_{\gamma_{k_1}}^{\gamma_{k_2}}(N-F)^{2j}F'\,dt=\sum_{k_1<k<k_2}\Psi_j(u_k)+\frac{(\frac12+u_{k_1})^{2j+1}+(\frac12-u_{k_2})^{2j+1}}{2j+1}$, where
\begin{equation}\label{eq:Psi}
\Psi_1(u)=u^2+\tfrac1{12},\qquad \Psi_2(u)=u^4+\tfrac12u^2+\tfrac1{80}.
\end{equation}
\end{theorem}

\section{The multiplicity lemma}\label{sec:lemma}
\begin{lemma}\label{lem:mult}
Let $F$ be any real function and $u_k$ as in \CGeqref{eq:defect}.
\begin{enumerate}[label=\textup{(\alph*)}]
\item If an ordinate $\gamma$ has multiplicity $m\ge2$ in the list $(\gamma_k)$, with indices $k,\dots,k+m-1$, then $u_{k+i}=u_k+i$ for $0\le i<m$, and for every $j\ge1$
\[
\textstyle\sum_{i=0}^{m-1}\bigl(2(u_k+i)\bigr)^{2j}\ \ge\ m .
\]
\item If $\gamma$ has multiplicity $1$, the zero with ordinate $\gamma$ is simple and lies on the critical line.
\end{enumerate}
\end{lemma}

\begin{proof}
(a) $F(\gamma_{k+i})=F(\gamma)$ for all $i$, so $u_{k+i}-u_k=i$. Put $\phi(x)=(2x)^{2j}$, even, convex, $\phi(\pm\frac12)=1$. For $m=2$, $|u|+|u+1|\ge1$ and convexity give $\phi(u)+\phi(u+1)\ge2\phi(\frac12)=2$. For $m\ge3$, at most one of the $m$ numbers $u+i$ lies in $(-\frac12,\frac12)$, and the others have absolute values at least $\frac12,\frac12,\frac32,\frac32,\dots$ in increasing order; hence the sum is at least $2+2\cdot3^{2j}\lfloor(m-2)/2\rfloor+(m-2-2\lfloor(m-2)/2\rfloor)3^{2j}\ge m$.
(b) If $\rho=\beta+i\gamma$ with $\beta\ne\frac12$, then $1-\bar\rho=1-\beta+i\gamma$ is another zero with the same ordinate; a multiple zero repeats its ordinate. In both cases the multiplicity of $\gamma$ is at least $2$.
\end{proof}

\section{Certificates from the defect}\label{sec:cert}
\begin{theorem}\label{thm:cert}
For every real function $F$, every $j\ge1$ and $T\ge2$,
\begin{equation}\label{eq:cert}
N_0^s(T,2T)\ \ge\ M_T-\sum_{k\in\mathcal I_T}(2u_k)^{2j}-O(\log T),
\end{equation}
with an absolute implied constant.
\end{theorem}

\begin{proof}
By Lemma~\CGref{lem:mult}(b) the zeros of $\mathcal I_T$ that are not counted by $N_0^s$ lie on ordinates of multiplicity $m\ge2$; by part (a) each such group of $m$ indices satisfies $m\le\sum_{i}(2u_{k+i})^{2j}$. Summing over the groups contained in $\mathcal I_T$ and adding the nonnegative terms of the other indices gives \CGeqref{eq:cert}; the groups that meet the ends of the window contain $O(\log T)$ zeros because $N(t+1)-N(t)\ll\log t$.
\end{proof}

\begin{corollary}\label{cor:Q}
For $F=F_X$ with $2\le X\le T$, as $T\to\infty$,
\begin{align}
\frac{N_0^s(T,2T)}{M_T}&\ \ge\ \frac43-4\,Q_2(T,X)-o(1),\label{eq:cert2}\\
\frac{N_0^s(T,2T)}{M_T}&\ \ge\ 1-16\,Q_4(T,X)+8\,Q_2(T,X)-\frac7{15}-o(1).\label{eq:cert4}
\end{align}
\end{corollary}

\begin{proof}
Apply Theorem~\CGref{thm:A} on $\mathcal I_T$ (the boundary terms and the change from $[\gamma_{k_1},\gamma_{k_2}]$ to $[T,2T]$ are $O((\log T)^{2j+1})=o(M_T)$, because $|N-F_X|\ll\log T+\sqrt X$ is not needed: the integrand is bounded on the two end pieces by $(\log T+\sigma_X)^{2j}(\mathcal L+M_X)$ and their length is $O(1)$; for $X\le T$ this is $o(M_T)$ only when $\sqrt X\,(\log T)^{2j}=o(T\log T)$, which holds). With \CGeqref{eq:Psi}, $\langle u^2\rangle=Q_2-\frac1{12}+o(1)$ and $\langle u^4\rangle=Q_4-\frac12\langle u^2\rangle-\frac1{80}+o(1)=Q_4-\frac12Q_2+\frac7{240}+o(1)$. Insert into \CGeqref{eq:cert} with $j=1$ and $j=2$.
\end{proof}

\begin{remark}
\CGeqref{eq:cert2} uses only the quadratic moment and is the form compared with pair correlation below. \CGeqref{eq:cert4} needs the quartic moment, which carries four-point information of the zeros; it is stronger numerically (Remark~\CGref{rem:experiments}).
\end{remark}

\section{Conditional form and thresholds}\label{sec:cond}
\begin{hypothesis}[$H(c,b)$]\label{hyp:Q}
For a fixed $0<c\le\frac12$ and a real $b$, $\limsup_{T\to\infty}Q_2(T,T^c)\le\frac1{2\pi^2}\bigl(\log\frac1c+b\bigr)$.
\end{hypothesis}

\begin{corollary}\label{cor:cond}
Under $H(c,b)$,
\begin{equation}\label{eq:condcert}
\liminf_{T\to\infty}\frac{N_0^s(T,2T)}{M_T}\ \ge\ \frac43-\frac2{\pi^2}\Bigl(\log\frac1c+b\Bigr).
\end{equation}
For $c=\frac12$ the right-hand side is $1.19285\ldots-0.20264\ldots\,b$.
\end{corollary}

\begin{proof}
\CGeqref{eq:cert2}.
\end{proof}

\begin{remark}\label{rem:thresholds}
The cross term $\frac1{M_T}\int(N-F_X)^2S_X'\,dt$ coming from $dF_X=(\mathcal L+S_X')\,dt$ is part of $Q_2$, hence of $b$; no density hypothesis on the zeros is needed. For $c=\frac12$: a positive proportion needs $b<5.887$; the proportion $\frac23$ needs $b<\frac{\pi^2}3-\log2=2.5967$; the value $0.6735$ reported in \cite{Pre261008965} needs $b<2.563$. By \cite[Thm.~B]{PCC11}, $Q_2(T,T^c)=\frac1{2\pi^2}\log\frac1c+O(1+\sqrt{\log(1/c)})$ unconditionally for $2\le T^c\le T^{1/2}$, with an absolute constant that is not computed.
\end{remark}

\section{Comparison with the 2026 results}\label{sec:comparison}
Unconditionally, more than two thirds of the zeros are simple and on the critical line \cite{AlpogeFurman2026,Lamzouri2026}, with the constant $0.67250\ldots$ of Montgomery and Taylor; both proofs use Montgomery's pair correlation in the unconditional form of \cite{BGSTB2024}, with Fourier support in $[-1,1]$, and an integrality step ($m^2\ge2m-1$ in the scalar form). Corollary~\CGref{cor:cond} uses a different integrality step (Lemma~\CGref{lem:mult}) and a different quantity, $Q_2$.

\begin{remark}\label{rem:extremal}
By \CGeqref{eq:cert2}, $N_0^s/M_T\le\frac23+o(1)$ forces $Q_2\ge\frac16-o(1)$, i.e. $b\ge\frac{\pi^2}3-\log2=2.5967$ in the normalisation of Hypothesis~\CGref{hyp:Q} at $c=\frac12$. In particular the extremal configurations of \cite[\S7.2]{AlpogeFurman2026} (two thirds simple zeros and the rest in doubles or shallow off-line pairs) have $b\ge2.5967$. The two certificates use disjoint information: the first, the form factor on $[-1,1]$; the second, the quantity $Q_2$, which by Heuristic~\CGref{heur:parseval} involves the form factor beyond $1$.
\end{remark}

\section{Heuristics and experiments}\label{sec:heur}
\begin{heuristic}\label{heur:parseval}
Since $S'=dN/dt-\mathcal L$, Parseval and Montgomery's form factor $F(\alpha)$ suggest
$Q_2(T,T^c)\approx\frac1{2\pi^2}\bigl(\log\frac1c+\int_1^\infty F(\alpha)\alpha^{-2}\,d\alpha\bigr)$, the part $c\le\alpha\le1$ giving $\log\frac1c$ through $F(\alpha)=\alpha+o(1)$. Under the pair correlation conjecture the integral is $1$, and Corollary~\CGref{cor:cond} would give $0.990$ at $c=\frac12$. This is not proved; the cross term of Remark~\CGref{rem:thresholds} is not accounted for by the heuristic.
\end{heuristic}

\begin{remark}[Experiments]\label{rem:experiments}
On Odlyzko's table of the first $2\,001\,052$ zeros, using only indices $k\le1\,400\,736$ (window $T=408\,392$, $M_T=745\,418$), the programs of the repository give the values of Table~\CGref{tab:exp}. On $T=10^5$, the cross term is $+0.0159$ of $Q_2=0.0950$ at $c=\frac12$ and $+0.0042$ of $0.1148$ at $c=\frac14$. An adaptation to $\zeta$ of the explicit Selberg constants of \cite{HiaryZhao2025} (not proved here) gives $b\le1.7\cdot10^7$, so the unconditional route through Selberg's method does not reach \CGeqref{eq:condcert} with $b<5.887$. Extending the certificate of \cite{Lamzouri2026} to kernels with support beyond $[-1,1]$ under an assumed bound $\int_1^\infty F\alpha^{-2}\le b$, $F\ge0$, a discretised optimisation gives at most $0.675$ for $b\ge1.5$; Corollary~\CGref{cor:cond} is larger for $b<2.597$.
\end{remark}
