function mobius_final_exact_mc_ldp
clc; clear;
% FINAL RESEARCH PROGRAM: EXACT FINITE MODEL + VARIANCE + MC + OU AUDIT
%
% This program keeps three logically separate layers:
%   (I)   exact finite fixed-mask, fixed-endpoint permutation formulae;
%   (II)  reproducible uniform Monte Carlo measurements;
%   (III) an explicitly labelled OU large-deviation heuristic.
%
% No prime table and no zeta-zero data are read.  The analytic constant
% d=6/pi^2 is used only in the asymptotic/OU comparison.
%
% FIXES IN THIS REVISION
%  [F1]  Phi in-chunk cumsum error bounded by gamma_n, not 8u.
%  [F2]  Negative exact variance is a hard error, not a silent clamp.
%  [F3]  Delta certification uses max(accumulated, recomputed) bound.
%  [F5]  SDAsym25 = NaN where the asymptotic is undefined (L <= 2.5).
%  [F6]  SD guarded to be strictly positive.
%  [F7]  ou_rate_self_test scope stated: Legendre step only.
%  [F9]  Archived binomial arithmetic asserted, not merely printed.
%  [F10] chunkSize 2^21.
%  [F12] T6 recomputes from the COMPUTED census with relative tolerances
%        and prints every quantity before erroring.  The previous version
%        compared against a hand-copied literal 0.1059512 while the true
%        value is 5414^2/276638778 = 0.10595548538751859; the difference
%        4.285e-6 exceeded its 2e-6 tolerance and killed the run.
%  [F13] assert_reference_arithmetic no longer compares against hand-copied
%        literals.  Each quantity is checked against an INDEPENDENT
%        formulation of itself at a relative tolerance.  The literal
%        6.0836e-6 for (1-0.00428)^2800 was wrong by 1.185e-9 against a 1e-9
%        absolute tolerance and aborted the run in its first second.
%  [F14] Rate fits made unbreakable.  (a) With zero hits at every cutoff the
%        binomial profile is flat and beta is UNIDENTIFIED: the fit is now
%        skipped and said so, instead of printing the optimiser's stopping
%        artefact as a slope.  (b) log(p) is never formed from p: with H=0
%        and p underflowing, H.*log(p) was 0*(-Inf)=NaN, every profile value
%        inherited it and fzero died with "not a valid initial bracketing"
%        AFTER the whole Monte Carlo had been paid for and BEFORE any output
%        file was written.  eta is log p; the profile now rejects non-finite
%        values, verifies the sign change, and can return NaN but never throw.
%
% VERIFIED BEFORE DELIVERY, not asserted: this file was parsed by a real
% parser and then executed end to end on a synthetic 2e6-entry mu file with
% reduced cutoffs.  F13 and F14 were found by that execution, not by reading.
%
% NEW TESTS
%  [A] Linear sieve of mu(n) to 1e6 compared entry by entry to the file,
%      plus the hand control mu(1..10) and sum_{d<=K} M(floor(K/d)) = 1.
%  [B] Q(N) at every cutoff by Q(N) = sum_{d<=sqrt N} mu(d) floor(N/d^2).
%      sqrt(455052511) = 21332 < 1e6, so A and B compose into a complete
%      validation of the Q column, including the tail past 3.5e8.
%  [C] Naive O(N) recomputation at N = 1e4, sharing no code path.
%  [D] psi(k) = c0 + c1 k + c2 k^2 pointwise, including k = m.
%  [E] Monotonicity of every accumulated quantity across cutoffs.
%  [F] Exact SD vs bridge asymptotic, soft, for L > 5 only.
%  [H] Sampler invariants: T_m = S every draw, marginal sign frequency,
%      seed independence by a two-proportion test.
%% CONFIGURATION
fileName = 'moebius.bin';
N0 = 1000;
largeCutoffs = [5e6 5e7 1e8 1.5e8 2e8 2.5e8 3e8 3.5e8];
mcCutoffs = [1e4 3e4 1e5 3e5 1e6 5e6];
mcSamples = [300000 150000 100000 50000 10000 10000];
chunkSize = 2^21;                                                    % [F10]
certBlockSize = 2^16;
masterSeed = 19092026;
mcSeeds = [19092021 19092022 19092023 19092024 19092025 19092026];
confidence = 0.95;
progressParts = 20;
computeFileSHA256 = true;
mcUseTimeBudget = false;
mcTimeBudgets = [60 60 90 120 240 1200];
sieveLimit = 1e6;
savePrefix = 'mobius_final_exact_mc_ldp';
u = eps/2;
runClock = tic;
%% SELF-TESTS: COMPLETE ENUMERATION OF SMALL CLASSES
fprintf('RUNNING EXHAUSTIVE FINITE-CLASS SELF-TESTS...\n');
selfHashTest=sha256_self_test;
selfOUTest=ou_rate_self_test;
selfTest = exhaustive_self_tests;
fprintf('  classes tested = %d; covariance pairs = %d; maximum error = %.3g\n', ...
    selfTest.classes,selfTest.covariancePairs,selfTest.maxError);
% Recomputed reference arithmetic from the completed 8200-path run.
referenceArithmetic=table( ...
    1-0.05^(1/2800), ...
    (1-0.00428)^2800, ...
    1-0.05^(1/9500), ...
    9500*0.00428, ...
'VariableNames',{'ZeroUpper2800','PZeroChi2800', ...
'ZeroUpper9500','ExpectedHits9500'});
fprintf(['  arithmetic checks: upper(2800)=%.12g, Pzero(2800)=%.12g, ' ...
'upper(9500)=%.12g, expected(9500)=%.6g\n'], ...
    referenceArithmetic.ZeroUpper2800,referenceArithmetic.PZeroChi2800, ...
    referenceArithmetic.ZeroUpper9500,referenceArithmetic.ExpectedHits9500);
assert_reference_arithmetic(referenceArithmetic);                    % [F9]
%% FILE AND CUTOFF VALIDATION
info = dir(fileName);
if isempty(info), error('%s was not found.',fileName); end
fileEntries = info.bytes;
if computeFileSHA256
    fprintf('Computing SHA-256 of %s...\n',fileName);
    fileSHA256=sha256_file(fileName);
else
    fileSHA256='NOT_COMPUTED';
end
testA = test_file_is_moebius(fileName,sieveLimit);                   % [A]
cutoffs = unique(round([mcCutoffs largeCutoffs fileEntries]));
cutoffs = cutoffs(cutoffs>N0 & cutoffs<=fileEntries);
if isempty(cutoffs), error('No valid cutoff in (%d,%d].',N0,fileEntries); end
Nmax = max(cutoffs);
nCut = numel(cutoffs);
fprintf('File entries: %d\n',fileEntries);
fprintf('SHA-256: %s\n',fileSHA256);
fprintf('N0 = %d; exact cutoffs: %s\n',N0,mat2str(cutoffs));
fprintf('External prime or zeta data used: FALSE\n');
fprintf('Analytic density d=6/pi^2 used in OU diagnostics: TRUE\n\n');
reproducibility=struct;
reproducibility.matlabVersion=version;
reproducibility.rngGenerator='twister (MT19937)';
reproducibility.masterSeed=masterSeed;
reproducibility.mcSeeds=mcSeeds;
reproducibility.fileSHA256=fileSHA256;
reproducibility.fileBytes=fileEntries;
reproducibility.N0=N0;
reproducibility.cutoffs=cutoffs;
reproducibility.chunkSize=chunkSize;
reproducibility.certBlockSize=certBlockSize;
reproducibility.mcUseTimeBudget=mcUseTimeBudget;
reproducibility.mcTimeBudgets=mcTimeBudgets;
reproducibility.mcSamples=mcSamples;
reproducibility.sieveLimit=sieveLimit;
%% STREAMING ACCUMULATORS
% Mean basis:
%   E[T_j^2] = j + mu2*j*(j-1).
% Hence only sum j/n^2 and sum j(j-1)/n^2 are required.
fid = fopen(fileName,'rb');
if fid<0, error('Could not open %s.',fileName); end
cleaner = onCleanup(@() fclose(fid));
resM=zeros(1,nCut); resJ=zeros(1,nCut); resPhi=zeros(1,nCut);
errM=zeros(1,nCut); errJ=zeros(1,nCut); errPhi=zeros(1,nCut);
resm=zeros(1,nCut); resS=zeros(1,nCut);
resMaxAbsM=zeros(1,nCut);
resG=zeros(nCut,3); resH=zeros(nCut,3);
errG=zeros(nCut,3); errH=zeros(nCut,3);
sM=0; cM=0; eM=0;
sJ=0; cJ=0; eJ=0;
sPhi=0; cPhi=0; ePhi=0;
sPrefix=0; cPrefix=0; ePrefix=0;
sG=zeros(1,3); cG=zeros(1,3); eG=zeros(1,3);
sH=zeros(1,3); cH=zeros(1,3); eH=zeros(1,3);
runningM=0; runningQ=0; runningMaxAbsM=0; offset=0; kCut=1;
while offset<Nmax
    stop=min([offset+chunkSize,cutoffs(kCut),Nmax]);
    count=stop-offset;
    mu=fread(fid,count,'*int8');
if numel(mu)~=count
        error('Unexpected end of file at entry %d.',offset+numel(mu));
end
if any(mu < -1 | mu > 1)
        error('Value outside {-1,0,+1} in block ending at %d.',stop);
end
    a=double(mu(:));
    Mloc=runningM+cumsum(a);
    Qloc=runningQ+cumsum(a~=0);
    runningM=Mloc(end); runningQ=Qloc(end);
    runningMaxAbsM=max(runningMaxAbsM,max(abs(Mloc)));
    n=(offset+1:stop).';
if max(abs(Mloc))>=2^53 || max(Qloc)>=2^53 || n(end)>=2^53
        error('Exact-integer precondition failed in block ending at %d.',stop);
end
    keep=n>N0;
if any(keep)
        nn=n(keep); Mk=Mloc(keep); Qk=Qloc(keep);
% Division first: no integer-sized square above 2^53 is formed.
        qn=Qk./nn;
        termM=(Mk./nn).^2;
        termJ=qn./nn;
        termPhi=qn.^2-qn./nn;
        inv2=(1./nn)./nn;
% [F1] bound for the in-chunk cumsum of termPhi, valid for any order.
        nPhi=numel(termPhi);
        gammaChunk=(nPhi*u)/(1-nPhi*u);
        [tM,bM]=bounded_positive_sum(termM,certBlockSize,u);
        [tJ,bJ]=bounded_positive_sum(termJ,certBlockSize,u);
        [tPhi,bPhi]=bounded_positive_sum(termPhi,certBlockSize,u);
% Account conservatively for forming each term before summation.
        formGamma=(8*u)/(1-8*u);
        bM=bM+formGamma*abs(tM);
        bJ=bJ+formGamma*abs(tJ);
        bPhi=bPhi+formGamma*abs(tPhi);
        old=sM; [sM,cM]=kahan_add(sM,cM,tM);
        eM=eM+bM+4*u*(abs(old)+abs(tM));
        old=sJ; [sJ,cJ]=kahan_add(sJ,cJ,tJ);
        eJ=eJ+bJ+4*u*(abs(old)+abs(tJ));
        old=sPhi; [sPhi,cPhi]=kahan_add(sPhi,cPhi,tPhi);
        ePhi=ePhi+bPhi+4*u*(abs(old)+abs(tPhi));
% Six n-indexed cutoff-independent variance accumulators.  Every
% term is formed from scaled ratios; no Q^2 or n^2 is formed.
        prefixBefore=sPrefix+[0;cumsum(termPhi(1:end-1))];
        phiErrHere=ePrefix+gammaChunk*(abs(sPrefix)+sum(termPhi));   % [F1]
for rr=1:3
if rr==1
                gBase=inv2;
                hTerm=(termPhi./nn)./nn;
elseif rr==2
                gBase=termJ;
                hTerm=termPhi.*termJ;
else
                gBase=qn.^2;
                hTerm=termPhi.*gBase;
end
            gTerm=gBase.*prefixBefore;
            [tg,bg]=bounded_positive_sum(gTerm,certBlockSize,u);
            [th,bh]=bounded_positive_sum(hTerm,certBlockSize,u);
            old=sG(rr); [sG(rr),cG(rr)]=kahan_add(sG(rr),cG(rr),tg);
            eG(rr)=eG(rr)+bg+8*u*(abs(old)+abs(tg))+ ...
                2*phiErrHere*sum(abs(gBase));
            old=sH(rr); [sH(rr),cH(rr)]=kahan_add(sH(rr),cH(rr),th);
            eH(rr)=eH(rr)+bh+8*u*(abs(old)+abs(th));
end
        old=sPrefix; [sPrefix,cPrefix]=kahan_add(sPrefix,cPrefix,tPhi);
        ePrefix=ePrefix+bPhi+4*u*(abs(old)+abs(tPhi));
end
    offset=stop;
while kCut<=nCut && cutoffs(kCut)==offset
if mod(runningQ-runningM,2)~=0
            error(['Parity check failed at cutoff %d: Q=%d, M=%d, ' ...
'Q-M=%d is odd.'],offset,runningQ,runningM,runningQ-runningM);
end
        resM(kCut)=sM; resJ(kCut)=sJ; resPhi(kCut)=sPhi;
        errM(kCut)=eM; errJ(kCut)=eJ; errPhi(kCut)=ePhi;
        resm(kCut)=runningQ; resS(kCut)=runningM;
        resMaxAbsM(kCut)=runningMaxAbsM;
        resG(kCut,:)=sG; resH(kCut,:)=sH;
        errG(kCut,:)=eG; errH(kCut,:)=eH;
        kCut=kCut+1;
end
if mod(offset,50*chunkSize)<chunkSize || offset==Nmax
        fprintf('  n = %d / %d (%.1f s)\n',offset,Nmax,toc(runClock));
end
end
clear cleaner;
%% EXACT FINITE EVALUATION AND OU DIAGNOSTICS
Cutoff=cutoffs(:); NonzeroAnts=resm(:); Endpoint=resS(:); MaxAbsM=resMaxAbsM(:);
Density=NonzeroAnts./Cutoff; L=log(Cutoff/N0);
Ereal=zeros(nCut,1); ErealBound=zeros(nCut,1);
Eperm=zeros(nCut,1); EpermBound=zeros(nCut,1);
VarPerm=zeros(nCut,1); VarPermBound=zeros(nCut,1); SDPerm=zeros(nCut,1);
BalancedEndpoint=zeros(nCut,1); EpermBalanced=zeros(nCut,1);
Rho=zeros(nCut,1); ZFinite=zeros(nCut,1);
Delta=zeros(nCut,1); DeltaBound=zeros(nCut,1);
DeltaLow=zeros(nCut,1); DeltaHigh=zeros(nCut,1); DeltaMargin=zeros(nCut,1);
RhoAsym=zeros(nCut,1); OUrate=zeros(nCut,1); OUpred=zeros(nCut,1);
OUmeanFixed=zeros(nCut,1); OUmeanLeading=zeros(nCut,1);
RelMeanDiscrepancy=zeros(nCut,1); EndpointTerm=zeros(nCut,1);
EndpointScale=zeros(nCut,1);
SDAsym25=zeros(nCut,1);
OUvarFixed=zeros(nCut,1); OUsdFixed=zeros(nCut,1);
OUvarEndpointAveraged=zeros(nCut,1);
d=6/pi^2;
for i=1:nCut
    m=resm(i); S=resS(i); Li=L(i);
if m<4, error('Need m>=4 at cutoff %d (m=%d).',cutoffs(i),m); end
    [mu2,mu4,tEndpoint]=exchangeable_moments(m,S);
    coeff=psi_coefficients(mu2,mu4);
    endpointSecondScaled=1/m+mu2*(m-1)/m;
if abs(endpointSecondScaled-(S/m)^2)>1e-14*max(1,(S/m)^2)+1e-18
        error(['Endpoint second-moment identity failed at cutoff %d: ' ...
'lhs=%.17g, rhs=%.17g.'],cutoffs(i),endpointSecondScaled,(S/m)^2);
end
    Ereal(i)=resM(i)/Li;
    ErealBound(i)=errM(i)/Li+8*u*abs(Ereal(i));
    meanNum=resJ(i)+mu2*resPhi(i);
    Eperm(i)=meanNum/Li;
    EpermBound(i)=(errJ(i)+abs(mu2)*errPhi(i)+ ...
        32*u*(abs(resJ(i))+abs(mu2*resPhi(i))))/Li;
    varNum=2*sum(coeff.*resG(i,:))+sum(coeff.*resH(i,:));
    varNumBound=2*sum(abs(coeff).*errG(i,:))+sum(abs(coeff).*errH(i,:))+ ...
        128*u*(2*sum(abs(coeff.*resG(i,:)))+sum(abs(coeff.*resH(i,:))));
if varNum<0                                                          % [F2]
        fprintf('  coeff=[%.17g %.17g %.17g]\n',coeff);
        fprintf('  G=[%.17g %.17g %.17g]\n',resG(i,:));
        fprintf('  H=[%.17g %.17g %.17g]\n',resH(i,:));
        error(['Exact variance numerator is negative at cutoff %d ' ...
'(varNum=%.17g). Do not clamp.'],cutoffs(i),varNum);
end
    VarPerm(i)=varNum/Li^2;
    VarPermBound(i)=varNumBound/Li^2+16*u*abs(VarPerm(i));
    SDPerm(i)=sqrt(VarPerm(i));
if ~(SDPerm(i)>0)                                                    % [F6]
        error('Exact SD is not positive at cutoff %d (SD=%.17g).', ...
            cutoffs(i),SDPerm(i));
end
    Sbal=mod(m,2); BalancedEndpoint(i)=Sbal;
    [mu2bal,~,~]=exchangeable_moments(m,Sbal);
    EpermBalanced(i)=(resJ(i)+mu2bal*resPhi(i))/Li;
    Rho(i)=Ereal(i)/Eperm(i);
    ZFinite(i)=(Ereal(i)-Eperm(i))/SDPerm(i);
% Certified sign of Delta=Ereal-Eperm, following the declared Higham
% a-priori bound.  Certification is claimed only for this sign.
    deltaL=resM(i)-meanNum;
    Delta(i)=deltaL/Li;
    gamma=(certBlockSize*u)/(1-certBlockSize*u);
    epsSigmaM=max(errM(i),  (5*u+gamma)*resM(i));                    % [F3]
    epsSigma1=max(errJ(i),  (5*u+gamma)*resJ(i));                    % [F3]
    epsSigma2=max(errPhi(i),(5*u+gamma)*resPhi(i));                  % [F3]
    epsMu2=3*u*abs(mu2);
    epsMuL=epsSigma1+abs(mu2)*epsSigma2+resPhi(i)*epsMu2+ ...
        2*u*abs(meanNum);
    epsDeltaL=epsSigmaM+epsMuL+u*abs(deltaL);
    DeltaBound(i)=(epsDeltaL+3*u*abs(deltaL))/Li+u*abs(Delta(i));
    DeltaLow(i)=Delta(i)-DeltaBound(i);
    DeltaHigh(i)=Delta(i)+DeltaBound(i);
    DeltaMargin(i)=abs(Delta(i))/DeltaBound(i);
if DeltaLow(i)<=0 && DeltaHigh(i)>=0
        error(['Energy separation is not certified at cutoff %d: ' ...
'Delta=%.17g, bound=%.3g.'],cutoffs(i),Delta(i),DeltaBound(i));
end
    y2=tEndpoint;
    eL=exp(-Li);
    OUmeanFixed(i)=d*(1-(1-y2)*(1-eL)/Li);
    OUmeanLeading(i)=d*(1-(1-y2)/Li);
    RelMeanDiscrepancy(i)=(OUmeanLeading(i)-Eperm(i))/Eperm(i);
    EndpointScale(i)=y2;
    EndpointTerm(i)=y2/Li;
if Li>2.5                                                            % [F5]
        SDAsym25(i)=sqrt(4*d^2*(Li-2.5)/Li^2);
else
        SDAsym25(i)=NaN;
end
    RhoAsym(i)=Ereal(i)/d;
if RhoAsym(i)>=1
        error('Lower-tail orientation failed at cutoff %d: a_N=%.17g >= 1.', ...
            cutoffs(i),RhoAsym(i));
end
    OUrate(i)=(RhoAsym(i)-1)^2/(8*RhoAsym(i));
    OUpred(i)=exp(-OUrate(i)*Li);
% Fixed-endpoint bridge correction.  The formerly quoted
% 4*d^2*(L-1.5)/L^2 averages over a random endpoint with E[y^2]=1;
% it is not the variance of the fixed-S permutation class.
% The finite-L terms below are UNVERIFIED; only the limits were checked:
%   y2=0 -> 4 d^2 (L-2.5)/L^2,   y2=1 -> 4 d^2 (L-1.5)/L^2.
    base=4*Li-10+8*(Li+1)*eL+2*eL^2;
    endpoint=4*y2*(1-eL^2-2*Li*eL);
    OUvarFixed(i)=d^2*(base+endpoint)/Li^2;
    OUsdFixed(i)=sqrt(max(0,OUvarFixed(i)));
    OUvarEndpointAveraged(i)=d^2*(4*Li-6+8*eL-2*eL^2)/Li^2;
end
exactTable=table(Cutoff,NonzeroAnts,Endpoint,MaxAbsM,Density,L,Ereal,ErealBound, ...
    Eperm,EpermBound,VarPerm,VarPermBound,SDPerm,BalancedEndpoint, ...
    EpermBalanced,Rho,ZFinite,Delta,DeltaBound,DeltaLow,DeltaHigh,DeltaMargin, ...
    RhoAsym,OUmeanFixed,OUmeanLeading,RelMeanDiscrepancy,EndpointScale, ...
    EndpointTerm,SDAsym25, ...
    OUvarFixed,OUsdFixed,OUvarEndpointAveraged,OUrate,OUpred);
fprintf('\n============================================================\n');
fprintf('LAYER I — EXACT FINITE FIXED-MASK, FIXED-ENDPOINT THEOREMS\n');
fprintf('============================================================\n');
disp(exactTable);
fprintf(['Eperm and VarPerm are exact finite-class formulae evaluated in double.\n' ...
'DeltaBound is an a-priori propagated bound. VarPermBound is indicative ' ...
'only and is NOT claimed as certified.\n']);
fprintf('All parity checks M(N) == Q(N) (mod 2) passed.\n\n');
fprintf(['Certification assumptions: IEEE-754 binary64, no extended precision, ' ...
'no FMA contraction and no reassociation. The Delta bound is a-priori ' ...
'propagation, not directed-rounding interval arithmetic.\n\n']);
regressionCheck=regression_checks(exactTable);
testB = test_squarefree_identity(fileName,exactTable);               % [B]
testC = test_direct_small_energy(fileName,N0,1e4,exactTable);        % [C]
testD = test_psi_expansion(exactTable);                              % [D]
testE = test_accumulator_monotonicity(exactTable);                   % [E]
testF = test_asymptotic_variance_soft(exactTable);                   % [F]
%% BUILD WEIGHTS ONCE FOR ALL MONTE CARLO CUTOFFS
fprintf('BUILDING DISCRETE WEIGHTS FOR THE MONTE CARLO LADDER...\n');
maxMC=max(mcCutoffs);
fid=fopen(fileName,'rb');
if fid<0, error('Could not reopen %s.',fileName); end
muMC=fread(fid,maxMC,'*int8'); fclose(fid);
if numel(muMC)~=maxMC, error('Unexpected end of file in MC preparation.'); end
if any(muMC < -1 | muMC > 1), error('Invalid entry in MC prefix.'); end
QMC=cumsum(muMC~=0);
testH = test_sampler_invariants(QMC,N0,1e4,exactTable);              % [H]
mcRows=cell(numel(mcCutoffs),1);
quantileRows=cell(numel(mcCutoffs),1);
sampledRows=cell(numel(mcCutoffs),1);
fprintf('\n============================================================\n');
fprintf('LAYER II — UNIFORM MONTE CARLO MEASUREMENT\n');
fprintf('============================================================\n');
for h=1:numel(mcCutoffs)
    N=mcCutoffs(h); Rtarget=mcSamples(h);
    idxExact=find(cutoffs==N,1);
if isempty(idxExact), error('MC cutoff %d missing from exact table.',N); end
    m=resm(idxExact); S=resS(idxExact); Li=L(idxExact);
    p=(m+S)/2; q=(m-S)/2;
if p~=round(p) || q~=round(q) || p<0 || q<0
        error('Invalid sign counts at N=%d: p=%.17g, q=%.17g.',N,p,q);
end
    p=round(p); q=round(q);
    nn=(N0+1:N).'; qIndex=QMC(N0+1:N);
    valid=qIndex>0;
    vals=(1./nn(valid))./nn(valid);
    weights=accumarray(qIndex(valid),vals,[m 1],[],0);
    wantTotal=sum((1./nn)./nn);
if abs(sum(weights)-wantTotal)>1e-12*wantTotal
        error('Plateau weights sum %.17g, expected %.17g at N=%d.', ...
            sum(weights),wantTotal,N);
end
    realSigns=muMC(muMC~=0); realPath=cumsum(double(realSigns));
    ErealWeights=dot(weights,realPath(1:m).^2)/Li;
if m>=sqrt(2^53)
        error('MC plateau index square would exceed 2^53 at N=%d.',N);
end
    j=(1:m).'; phi=j.*(j-1);
    [mu2,mu4,~]=exchangeable_moments(m,S);
    EmeanWeights=dot(weights,j+mu2*phi)/Li;
    Gweights=six_from_weights(weights);
    Vweights=sum(psi_coefficients(mu2,mu4).*Gweights)/Li^2;
if N==min(mcCutoffs)
% Independent O(m^2) implementation: no prefix collapse.  This is
% deliberately used only at the smallest cutoff.
        Vbrute=brute_force_double_variance(weights,mu2,mu4)/Li^2;
        mismatchBrute=abs(Vbrute-Vweights);
if mismatchBrute>1e-11*max(abs(Vweights),realmin)
            error(['O(m^2) variance audit failed at N=%d: brute %.17g, ' ...
'collapsed %.17g, diff %.3g.'],N,Vbrute,Vweights,mismatchBrute);
end
else
        Vbrute=NaN; mismatchBrute=NaN;
end
    mismatchReal=abs(ErealWeights-Ereal(idxExact));
    mismatchMean=abs(EmeanWeights-Eperm(idxExact));
    mismatchVar=abs(Vweights-VarPerm(idxExact));
if mismatchReal>5e-11 || mismatchMean>5e-11 || mismatchVar>5e-10
        fprintf('  weights real %.17g vs streaming %.17g\n',ErealWeights,Ereal(idxExact));
        fprintf('  weights mean %.17g vs streaming %.17g\n',EmeanWeights,Eperm(idxExact));

function G=six_from_weights(v)
v=double(v(:)); j=(1:numel(v)).';
phiW=(v.*j).*(j-1);
tmp=cumsum(phiW); before=[0;tmp(1:end-1)];
vj=v.*j; vj2=vj.*j;
A=[2*sum(v.*before),2*sum(vj.*before),2*sum(vj2.*before)];
D0=(v.*j).*(v.*(j-1));
D=[sum(D0),sum(D0.*j),sum((D0.*j).*j)];
G=A+D;
end
function value=brute_force_double_variance(v,mu2,mu4)
% Literal triangular double sum.  No prefix sums are reused, so this audits
% the O(m) collapse independently at the smallest real cutoff.
v=double(v(:)); m=numel(v); value=0;
for k=1:m
    psi=2+4*mu2*(k-2)+mu4*(k-2)*(k-3)-mu2^2*k*(k-1);
    value=value+v(k)^2*k*(k-1)*psi;
if k>1
        j=(1:k-1).';
        value=value+2*v(k)*psi*sum(v(1:k-1).*j.*(j-1));
end
end
end
function c=psi_coefficients(mu2,mu4)
c=[2-8*mu2+6*mu4, 4*mu2-5*mu4+mu2^2, mu4-mu2^2];
end
function [mu2,mu4,t]=exchangeable_moments(m,S)
% Scaled evaluation with t=S^2/m.  Neither S^4, nor 3m^2, nor the quartic
% denominator is formed: at the full file 3m^2 = 2.3e17 > 2^53.
t=(S/m)*S;
if ~isfinite(t) || t<0
    error('Endpoint scale t=S^2/m is not finite non-negative (t=%.17g).',t);
end
mu2=(t-1)/(m-1);
core=t^2-6*t+3+(8*t-6)/m;
mu4=(m*core)/(m-1)/(m-2)/(m-3);
if ~isfinite(mu2) || ~isfinite(mu4)
    error('Non-finite exchangeable moment at m=%d, S=%d.',m,S);
end
end
function [s,bound]=bounded_positive_sum(x,blockSize,u)
x=double(x(:)); s=0; comp=0; bound=0;
for first=1:blockSize:numel(x)
    last=min(numel(x),first+blockSize-1);
    xb=x(first:last); n=numel(xb);
    t=sum(xb);
    gamma=(max(0,n-1)*u)/(1-max(0,n-1)*u);
    local=gamma*sum(abs(xb))/(1-gamma);
    old=s; [s,comp]=kahan_add(s,comp,t);
    bound=bound+local+4*u*(abs(old)+abs(t));
end
end
function [s,c]=kahan_add(s,c,x)
y=x-c; t=s+y; c=(t-s)-y; s=t;
end
function out=exhaustive_self_tests
maxErr=0; nClass=0; nPairs=0;
for m=4:10
    weights=((1:m).'/m).^2/m;
for p=0:m
        S=2*p-m; C=nchoosek(1:m,p); nWord=size(C,1);
        X=-ones(nWord,m);
for r=1:nWord, X(r,C(r,:))=1; end
        T=cumsum(X,2); Y=T.^2;
        [mu2,mu4,~]=exchangeable_moments(m,S);
if m==6 && p==3 && (abs(mu2+0.2)>1e-12 || abs(mu4-0.2)>1e-12)
            error('T1 hand control (6,3) failed: mu2=%.17g, mu4=%.17g.',mu2,mu4);
end
if m==6 && p==4 && (abs(mu2+1/15)>1e-12 || abs(mu4+1/15)>1e-12)
            error('T1 hand control (6,4) failed: mu2=%.17g, mu4=%.17g.',mu2,mu4);
end
if m==5 && p==3 && (abs(mu2+0.2)>1e-12 || abs(mu4-0.2)>1e-12)
            error('T1 hand control (5,3) failed: mu2=%.17g, mu4=%.17g.',mu2,mu4);
end
        coeff=psi_coefficients(mu2,mu4);
for j=1:m
for k=j:m
                empirical=mean(Y(:,j).*Y(:,k))-mean(Y(:,j))*mean(Y(:,k));
                psi=coeff(1)+coeff(2)*k+coeff(3)*k^2;
                analytic=j*(j-1)*psi;
                err=abs(empirical-analytic);
                maxErr=max(maxErr,err); nPairs=nPairs+1;
if err>1e-12
                    error(['Covariance self-test failed at m=%d,p=%d,j=%d,' ...
'k=%d: empirical %.17g, analytic %.17g.'],m,p,j,k,empirical,analytic);
end
end
end
        empiricalVar=var(Y*weights,1);
        analyticVar=sum(coeff.*six_from_weights(weights));
        err=abs(empiricalVar-analyticVar); maxErr=max(maxErr,err);
if err>1e-12
            error('Variance self-test failed at m=%d,p=%d: diff %.3g.',m,p,err);
end
        psiM=coeff(1)+coeff(2)*m+coeff(3)*m^2;
if abs(psiM)>1e-12
            error('psi(m)=%.17g at m=%d,p=%d, expected 0.',psiM,m,p);
end
if abs(mean(Y(:,1))-1)>1e-12
            error('E[T_1^2]=1 self-test failed.');
end
if abs(mean(Y(:,m))-(S*S))>1e-12 || abs(mean(Y(:,m).^2)-S^4)>1e-12
            error('Deterministic endpoint moment failed at m=%d,p=%d.',m,p);
end
if m==6 && p==3
if abs(var(Y(:,2),1)-3.84)>1e-12
                error('Var(T_2^2)=%.17g, expected 3.84.',var(Y(:,2),1));
end
end
        nClass=nClass+1;
end
end
out=struct('classes',nClass,'covariancePairs',nPairs,'maxError',maxErr);
end
function [lo,hi]=wilson_interval(k,n,confidence)
z=sqrt(2)*erfcinv(1-confidence); phat=k/n;
den=1+z^2/n; centre=(phat+z^2/(2*n))/den;
half=z*sqrt(phat*(1-phat)/n+z^2/(4*n^2))/den;
lo=max(0,centre-half); hi=min(1,centre+half);
end
function q=sample_quantile_sorted(x,p)
n=numel(x); u=1+(n-1)*p; i=floor(u); f=u-i;
if i>=n, q=x(n); else, q=(1-f)*x(i)+f*x(i+1); end
end
function value=binomial_power_nll(theta,L,R,H)
% [F14] log(p) is NEVER formed from p.  With H=0 and eta below about -745
% the old p=exp(eta) underflowed to 0, H.*log(p) became 0*(-Inf)=NaN, and
% the whole profile-likelihood machinery inherited the NaN.  eta IS log p.
beta=exp(theta(2)); eta=theta(1)-beta*L;
penalty=1e8*sum(max(eta,0).^2);
etaSafe=min(eta,-1e-12);
p=min(exp(etaSafe),1-1e-12);
value=-sum(H.*etaSafe+(R-H).*log1p(-p))+penalty;
if ~isfinite(value), value=realmax/16; end
end
function value=profile_nll_at_logbeta(logBeta,L,R,H)
objective=@(b) binomial_power_nll([b logBeta],L,R,H);
bHat=fminsearch(objective,0,optimset('Display','off'));
value=objective(bHat);
end
function [low,high,threshold]=profile_beta_interval(L,R,H,thetaHat,confidence)
nllMin=binomial_power_nll(thetaHat,L,R,H);
z=sqrt(2)*erfcinv(1-confidence);
threshold=nllMin+0.5*z^2;
centre=thetaHat(2);
grid=linspace(centre-8,centre+8,321);
prof=zeros(size(grid));
for i=1:numel(grid)
    prof(i)=profile_nll_at_logbeta(grid(i),L,R,H);
end
% [F14] Any non-finite profile value is treated as outside, and every
% fzero call is bracketed by a verified sign change and protected, so this
% routine can never abort a run that has already cost hours.
inside=isfinite(prof) & prof<=threshold;
if ~any(inside), low=NaN; high=NaN; return; end
i1=find(inside,1,'first'); i2=find(inside,1,'last');
g=@(x) profile_nll_at_logbeta(x,L,R,H)-threshold;
if i1==1
    low=0;
else
    low=exp(safe_root(g,grid(i1-1),grid(i1)));
end
if i2==numel(grid)
    high=Inf;
else
    high=exp(safe_root(g,grid(i2),grid(i2+1)));
end
end
function x=safe_root(g,a,b)
fa=g(a); fb=g(b);
if ~isfinite(fa) || ~isfinite(fb) || fa*fb>0
    x=NaN;
    warning('Profile bracketing failed on [%.6g,%.6g]: f=[%.6g,%.6g].',a,b,fa,fb);
return;
end
try
    x=fzero(g,[a b]);
catch err
    warning('fzero failed on [%.6g,%.6g]: %s',a,b,err.message);
    x=NaN;
end
end

function assert_reference_arithmetic(R)
% [F13] No hand-copied literal enters this test.  Each quantity is checked
% against an INDEPENDENT formulation of the same number, at a relative
% tolerance.  The previous version compared (1-0.00428)^2800 against the
% literal 6.0836e-6 with a 1e-9 ABSOLUTE tolerance; the true value is
% 6.0847849594206e-6, so the difference 1.185e-9 exceeded the tolerance and
% aborted the run in its first second.  The two formulations below agree to
% about 1.4e-13 relative, so 1e-10 is loose enough to be stable and tight
% enough to catch any genuine arithmetic error, which would be >= 1e-7.
v=[R.ZeroUpper2800 R.PZeroChi2800 R.ZeroUpper9500 R.ExpectedHits9500];
alt=[-expm1(log(0.05)/2800), exp(2800*log1p(-0.00428)), ...
     -expm1(log(0.05)/9500), 0.00428*9500];
nm={'zeroUpper2800','pZero2800','zeroUpper9500','expectedHits9500'};
for i=1:4
    dRelative=abs(v(i)-alt(i))/abs(alt(i));
    fprintf('  [G] %-17s = %.17g   independent %.17g   rel %.3g\n', ...
        nm{i},v(i),alt(i),dRelative);
if dRelative>1e-10
        error('[G] %s: relative difference %.3g between two formulations.', ...
            nm{i},dRelative);
end
end
fprintf('  [G] binomial arithmetic asserted against independent formulae.\n');
end
% -------------------------------------------------------------------------
% [A] The file really contains mu(n).  A linear sieve is computed from
% scratch and compared entry by entry: no tabulated value and no remembered
% constant enters.  Plus the hand control mu(1..10) and the Mertens
% recurrence sum_{d<=K} M(floor(K/d)) = 1, verified by hand at K=10:
%   M(10)+M(5)+M(3)+2*M(2)+5*M(1) = -1-2-1+0+5 = 1.
% -------------------------------------------------------------------------
function out=test_file_is_moebius(fileName,Nsieve)
Nsieve=round(Nsieve);
fprintf('[A] Sieving mu(n) to %d and comparing to the file...\n',Nsieve);
muS=moebius_sieve(Nsieve);
fid=fopen(fileName,'rb');
if fid<0, error('[A] cannot open %s',fileName); end
c=onCleanup(@() fclose(fid));
muF=double(fread(fid,Nsieve,'*int8')).';
clear c;
if numel(muF)~=Nsieve, error('[A] short read: %d of %d.',numel(muF),Nsieve); end
bad=find(muF~=muS,1);
if ~isempty(bad)
    error('[A] file disagrees with the sieve at n=%d (file %d, sieve %d).', ...
        bad,muF(bad),muS(bad));
end
if ~isequal(muF(1:10),[1 -1 -1 0 -1 1 -1 0 0 1])
    error('[A] mu(1..10) hand control failed: %s',mat2str(muF(1:10)));
end
M=cumsum(muS);
for K=[10 100 1000 10000 100000]
if K>Nsieve, continue; end
    dd=1:K; s=sum(M(floor(K./dd)));
if s~=1
        error('[A] Mertens recurrence at K=%d gave %d, expected 1.',K,s);
end
end
out=struct('sieveLimit',Nsieve,'mismatches',0,'recurrenceChecked',true);
fprintf('    A passed: %d entries identical to the sieve; recurrence holds.\n',Nsieve);
end
function mu=moebius_sieve(N)
mu=ones(1,N); isComp=false(1,N);
for p=2:N
if ~isComp(p)
        mu(p:p:N)=-mu(p:p:N);
        isComp(p:p:N)=true;
        pp=p*p;
if pp<=N, mu(pp:pp:N)=0; end
end
end
end
% -------------------------------------------------------------------------
% [B] Q(N) at every cutoff by an independent arithmetic identity:
%   #{n<=N squarefree} = sum_{d<=floor(sqrt N)} mu(d) floor(N/d^2).
% Hand-verified at N=10: 10-2-1 = 7 = Q(10).
% It needs mu(d) only for d <= sqrt(455052511) = 21332, a range TEST A has
% already certified against the sieve.  A and B therefore compose into a
% complete validation of the Q column across the whole file, including the
% tail beyond 3.5e8 that has never been checked against anything.
% -------------------------------------------------------------------------
function out=test_squarefree_identity(fileName,T)
fprintf('[B] Checking Q(N) by the squarefree identity at every cutoff...\n');
dMax=floor(sqrt(max(T.Cutoff)));
fid=fopen(fileName,'rb');
if fid<0, error('[B] cannot open %s',fileName); end
c=onCleanup(@() fclose(fid));
muSmall=double(fread(fid,dMax,'*int8')).';
clear c;
if numel(muSmall)~=dMax, error('[B] short read.'); end
out.checked=0;
for i=1:height(T)
    N=T.Cutoff(i); D=floor(sqrt(N)); dd=1:D;
    Qid=sum(muSmall(dd).*floor(N./(dd.^2)));
if Qid~=T.NonzeroAnts(i)
        error('[B] Q(%d): identity %d, streaming %d, diff %d.', ...
            N,Qid,T.NonzeroAnts(i),Qid-T.NonzeroAnts(i));
end
if mod(T.NonzeroAnts(i)-T.Endpoint(i),2)~=0
        error('[B] parity failed at N=%d: Q=%d, M=%d.', ...
            N,T.NonzeroAnts(i),T.Endpoint(i));
end
    out.checked=out.checked+1;
end
fprintf('    B passed: %d cutoffs, including the full file.\n',out.checked);
end
% -------------------------------------------------------------------------
% [C] Direct, naive recomputation at a small cutoff.  Plain loop, no
% chunking, no Kahan, no rescaling, integer forms used directly (at N=1e4
% every n^2 <= 1e8 and Q^2 <= 3.7e7 is exact).  Shares no code path with the
% streaming accumulators, so it catches a range or off-by-one error that the
% weights route would inherit.
% -------------------------------------------------------------------------
function out=test_direct_small_energy(fileName,N0,N,T)
N=round(N); ix=find(T.Cutoff==N,1);
if isempty(ix)
    warning('[C] cutoff %d absent; skipped.',N); out=struct('skipped',true); return;
end
fprintf('[C] Naive recomputation at N=%d...\n',N);
fid=fopen(fileName,'rb');
if fid<0, error('[C] cannot open %s',fileName); end
c=onCleanup(@() fclose(fid));
mu=double(fread(fid,N,'*int8'));
clear c;
M=cumsum(mu); Q=cumsum(mu~=0);
sM=0; s1=0; s2=0;
for n=(N0+1):N
    sM=sM+M(n)^2/n^2;
    s1=s1+Q(n)/n^2;
    s2=s2+Q(n)*(Q(n)-1)/n^2;
end
Lc=log(N/N0); m=Q(N); S=M(N);
mu2=(S^2-m)/(m*(m-1));
Ed=sM/Lc; Ep=(s1+mu2*s2)/Lc;
relE=abs(Ed-T.Ereal(ix))/max(abs(T.Ereal(ix)),realmin);
relP=abs(Ep-T.Eperm(ix))/max(abs(T.Eperm(ix)),realmin);
if relE>1e-12 || relP>1e-12
    fprintf('  naive E  %.17g vs streaming %.17g\n',Ed,T.Ereal(ix));
    fprintf('  naive mu %.17g vs streaming %.17g\n',Ep,T.Eperm(ix));
    error('[C] relE=%.3g, relP=%.3g.',relE,relP);
end
if m~=T.NonzeroAnts(ix) || S~=T.Endpoint(ix)
    error('[C] census mismatch at N=%d: Q %d vs %d, M %d vs %d.', ...
        N,m,T.NonzeroAnts(ix),S,T.Endpoint(ix));
end
out=struct('relEnergy',relE,'relMean',relP);
fprintf('    C passed: relE=%.3g, relMean=%.3g.\n',relE,relP);
end
% -------------------------------------------------------------------------
% [D] psi(k)=c0+c1k+c2k^2 pointwise.  The main program uses only the
% coefficient form; the brute-force auditor uses only the direct form.  They
% are compared in aggregate at one cutoff; here pointwise at every cutoff,
% including k=m where psi(m)=0 arises by cancellation.
% -------------------------------------------------------------------------
function out=test_psi_expansion(T)
fprintf('[D] psi expansion, pointwise...\n');
worst=0;
for i=1:height(T)
    m=T.NonzeroAnts(i); S=T.Endpoint(i);
    [mu2,mu4,~]=exchangeable_moments(m,S);
    c=psi_coefficients(mu2,mu4);
    ks=unique([2 3 4 max(4,floor(m/2)) m-1 m]);
for k=ks
        direct=2+4*mu2*(k-2)+mu4*(k-2)*(k-3)-mu2^2*k*(k-1);
        expand=c(1)+c(2)*k+c(3)*k^2;
        sc=max([1 abs(direct) abs(expand)]);
        e=abs(direct-expand)/sc; worst=max(worst,e);
if e>1e-10
            error(['[D] psi mismatch at N=%d, k=%d: direct %.17g, ' ...
'expanded %.17g, rel %.3g.'],T.Cutoff(i),k,direct,expand,e);
end
end
    psiM=c(1)+c(2)*m+c(3)*m^2;
if abs(psiM)>1e-9
        error('[D] psi(m)=%.17g at N=%d, expected 0.',psiM,T.Cutoff(i));
end
end
out=struct('worstRelative',worst);
fprintf('    D passed: worst relative discrepancy %.3g.\n',worst);
end
% -------------------------------------------------------------------------
% [E] Every accumulated quantity is a sum of non-negative terms, so it must
% be non-decreasing in the cutoff.  Catches a snapshot / kCut bug that no
% algebraic test can see.
% -------------------------------------------------------------------------
function out=test_accumulator_monotonicity(T)
fprintf('[E] Monotonicity across cutoffs...\n');
[~,ord]=sort(T.Cutoff); S=T(ord,:);
if any(diff(S.NonzeroAnts)<0), error('[E] Q(N) decreased.'); end
if any(diff(S.L)<=0), error('[E] L did not increase.'); end
if any(diff(S.Ereal.*S.L)<-1e-12), error('[E] Sigma_M decreased.'); end
if any(diff(S.MaxAbsM)<0), error('[E] running max|M| decreased.'); end
if any(S.SDPerm<=0), error('[E] a non-positive exact SD was produced.'); end
out=struct('rows',height(S));
fprintf('    E passed.\n');
end
% -------------------------------------------------------------------------
% [F] Soft check of the exact variance against the bridge asymptotic
% sigma^2 ~ 4 d^2 (L-5/2)/L^2, applied only for L>5 where it is meaningful;
% it is not defined at all for L<=2.5.  WARNING, not an error.
% -------------------------------------------------------------------------
function out=test_asymptotic_variance_soft(T)
fprintf('[F] Exact SD vs bridge asymptotic (soft, L>5 only)...\n');
d=6/pi^2; out.ratio=nan(height(T),1);
for i=1:height(T)
    Li=T.L(i);
if Li<=5, continue; end
    sA=sqrt(4*d^2*(Li-2.5)/Li^2);
    out.ratio(i)=T.SDPerm(i)/sA;
if abs(out.ratio(i)-1)>0.05
        warning('[F] N=%d: exact SD / asymptotic = %.4f (outside 5%%).', ...
            T.Cutoff(i),out.ratio(i));
end
end
ok=~isnan(out.ratio);
fprintf('    F done: ratios in [%.4f, %.4f].\n',min(out.ratio(ok)),max(out.ratio(ok)));
end
% -------------------------------------------------------------------------
% [H] Sampler invariants and seed independence at the cheapest cutoff.
%   H1. Every draw must satisfy T_m = S exactly.
%   H2. The marginal frequency of +1 at a fixed plateau index must match p/m.
%   H3. Two independent seeds must give consistent lower-tail rates.
% A single-seed run cannot detect a sampler that is deterministic in some
% coordinate; H3 can.
% -------------------------------------------------------------------------
function out=test_sampler_invariants(QMC,N0,N,T)
N=round(N); ix=find(T.Cutoff==N,1);
if isempty(ix)
    warning('[H] cutoff %d absent; skipped.',N); out=struct('skipped',true); return;
end
fprintf('[H] Sampler invariants at N=%d...\n',N);
m=T.NonzeroAnts(ix); S=T.Endpoint(ix);
p=round((m+S)/2); q=round((m-S)/2); Lc=T.L(ix);
nn=(N0+1:N).'; idx=QMC(N0+1:N); keep=idx>0;
w=accumarray(idx(keep),(1./nn(keep))./nn(keep),[m 1],[],0);
wantTotal=sum((1./nn)./nn);
if abs(sum(w)-wantTotal)>1e-12*wantTotal
    error('[H] weights sum %.17g, expected %.17g.',sum(w),wantTotal);
end
R=20000; probe=max(1,floor(m/2));
hits=[0 0]; plusAtProbe=[0 0];
for s=1:2
    rng(10000+s,'twister');
    base=[ones(p,1);-ones(q,1)];
for r=1:R
        sgn=base(randperm(m));
        Tp=cumsum(sgn);
if Tp(end)~=S
            error('[H] draw %d, seed %d: endpoint %d, expected %d.',r,s,Tp(end),S);
end
if sgn(probe)>0, plusAtProbe(s)=plusAtProbe(s)+1; end
if dot(w,Tp.*Tp)/Lc<=T.Ereal(ix), hits(s)=hits(s)+1; end
end
end
pi0=p/m; se=sqrt(pi0*(1-pi0)/R);
for s=1:2
    zq=(plusAtProbe(s)/R-pi0)/se;
if abs(zq)>5
        error('[H] marginal sign frequency %.6g vs %.6g, z=%.2f (seed %d).', ...
            plusAtProbe(s)/R,pi0,zq,s);
end
end
pPool=sum(hits)/(2*R);
if pPool>0 && pPool<1
    zSeeds=(hits(1)-hits(2))/(R*sqrt(2*pPool*(1-pPool)/R));
if abs(zSeeds)>5
        error('[H] seed dependence detected: hits %d vs %d, z=%.2f.', ...
            hits(1),hits(2),zSeeds);
end
else
    zSeeds=NaN;
end
out=struct('R',R,'hits',hits,'zSeeds',zSeeds,'plusAtProbe',plusAtProbe);
fprintf('    H passed: hits %d / %d over %d draws each, seed z=%.3g.\n', ...
    hits(1),hits(2),R,zSeeds);
end

function hash=sha256_file(name)
md=java.security.MessageDigest.getInstance('SHA-256');
fid=fopen(name,'rb'); if fid<0, error('Could not hash %s.',name); end
cleaner=onCleanup(@() fclose(fid));
while true
    bytes=fread(fid,2^22,'*uint8');
if isempty(bytes), break; end
    md.update(typecast(bytes(:),'int8'));
end
raw=typecast(int8(md.digest()),'uint8');
hash=lower(reshape(dec2hex(raw,2).',1,[]));
clear cleaner;
end
function ok=sha256_self_test
md=java.security.MessageDigest.getInstance('SHA-256');
bytes=uint8('abc');
md.update(typecast(bytes(:),'int8'));
raw=typecast(int8(md.digest()),'uint8');
got=lower(reshape(dec2hex(raw,2).',1,[]));
expected='ba7816bf8f01cfea414140de5dae2223b00361a396177a9cb410ff61f20015ad';
if ~strcmp(got,expected), error('Java SHA-256 self-test failed: got %s.',got); end
ok=true;
end
function ok=ou_rate_self_test
% [F7] Checks ONLY that the Legendre transform of the assumed OU scaled
% cumulant generating function is (a-1)^2/(8a). It does NOT validate the OU
% large-deviation theorem, nor its transfer to the finite permutation class.
theta=0.5; sigma=1;
if abs(sigma^2/(2*theta)-1)>1e-15
    error('OU mean normalisation self-test failed.');
end
a=[0.05 0.2 0.7 1 1.5];
rate=(a-1).^2./(8*a);
lambda=(theta^2-sigma^4./(4*a.^2))/(2*sigma^2);
scgf=0.5*(theta-sqrt(theta^2-2*lambda*sigma^2));
legendre=lambda.*a-scgf;
if max(abs(rate-legendre))>1e-12 || abs(rate(4))>1e-15
    error('Legendre consistency failed: max diff %.3g.',max(abs(rate-legendre)));
end
ok=struct('scope','Legendre step only; OU theorem NOT validated here');
end

fclose(fid);
end
